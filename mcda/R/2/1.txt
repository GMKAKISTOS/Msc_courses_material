#
# Sample distributions, sample mean and sample variance, for Normal r.v.
#

M<-1000
x1<-rnorm(M)
x2<-rnorm(M)
x3<-rnorm(M)
x4<-rnorm(M)
x<-data.frame(x1,x2,x3,x4)
x

apply(x,1,mean)
apply(x,2,mean)

Z<-apply(x,1,mean) # sample mean from 4 observations Z=(x1+x2+x3+x4)/4
                   # (x1+x2+x3+x4)/4 ~ N(E(x1),var(x1)/4)
summary(Z) 
mean(Z)            # kovta sto 0
var(Z)             # kovta sto 1/4

hist(Z,breaks=50,probability=TRUE)
curve(dnorm(x,mean=0,sd=1/sqrt(4)),add=TRUE,col=3,lwd=2)


apply(x,2,mean)    # afora xwrista tis x1, x2, x3, x4. Kovta sto 0
apply(x,2,var)     # Kovta sto 1


S2<-apply(x,1,var) # sample variance from 4 observations 
                   # S^2=[(x1-Z)^2+(x2-Z)^2+(x3-Z)^2+(x4-Z)^2]/[4-1]
                   # [4-1]*S^2 ~ X^2 with 3 df that is Gamma(3/2,2)

summary(3*S2)
mean(3*S2)        # kovta sto 3/2*2=3
var(3*S2)         # kovta sto (3/2)*2^2=6

hist(3*S2,breaks=50,probability=TRUE)
curve(dchisq(x,df=3),add=TRUE,col=3,lwd=2)

#
# Creat normal observations from uniform observations
# 
# Box-Muller method 
# 
# for this part use the online compiler 
# 
# https://rdrr.io/snippets/
#
# since for the plot the library(ggplot2) and library(ggExtra)
# are required

M<-1000
U1<-runif(M)
U2<-runif(M)

R<-sqrt(-2*log(U1))
Theta<-2*pi*U2
Z1<-R*cos(Theta)
Z2<-R*sin(Theta)
Z<-data.frame(Z1,Z2)

cov(Z)
cor(Z)    # Z1, Z2 are independent

library(ggplot2)
library(ggExtra)

p<-ggplot(Z, aes(x=Z1, y=Z2))+
  geom_point() +
  theme_classic()

ggMarginal(p, type="histogram",bins=30,xparams=list(colour="blue",fill="red"))
